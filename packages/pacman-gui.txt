# pacman-gui.txt — Desktop apps and fonts from the official Arch repos.
#
# Linux counterparts to the `cask` section of brew/Brewfile. Kept separate from
# pacman.txt so a headless box can skip all of it with `bootstrap.sh --skip-gui`.
#
# Every name here was checked against the official repos; anything that only
# exists in the AUR is handled by cachyos-extra.sh instead.

# --- Applications ---
ghostty       # cask "ghostty"
tailscale     # cask "tailscale-app" (CLI + daemon; no tray app on Linux)
moonlight-qt  # cask "moonlight"
rclone        # cask "google-drive" — Google has no Linux client; rclone syncs Drive
